Custom Cursor & Micro-Interaction Playground

A gallery of hover/cursor micro-interactions, each a standalone, copy-pasteable snippet. Every effect below is dependency-free vanilla JS/CSS, except the damped cursor-follower, which uses GSAP’s quickTo — the field-standard tool for that specific kind of inertia-based animation.

1. Damped cursor-follow dot

Move your mouse anywhere on this page. A dot + ring trail it with GSAP-driven inertia. Disabled automatically on touch devices and under reduced motion.

(watch the cursor, not this box)
View source 
const dotX = gsap.quickTo(dot, 'x', { duration: 0.1, ease: 'power3' });
const dotY = gsap.quickTo(dot, 'y', { duration: 0.1, ease: 'power3' });
const ringX = gsap.quickTo(ring, 'x', { duration: 0.5, ease: 'power3' });
const ringY = gsap.quickTo(ring, 'y', { duration: 0.5, ease: 'power3' });

document.addEventListener('mousemove', (e) => {
  dotX(e.clientX);
  dotY(e.clientY);
  ringX(e.clientX);
  ringY(e.clientY);
});

2. Magnetic button

Hover near the button — it nudges toward your cursor, then springs back.

View source 
function magneticOffset(cursorPos, elementCenter, radius, strength) {
  const dx = cursorPos.x - elementCenter.x;
  const dy = cursorPos.y - elementCenter.y;
  const distance = Math.hypot(dx, dy);
  if (radius <= 0 || distance > radius) return { x: 0, y: 0 };
  return { x: dx * strength, y: dy * strength };
}

button.addEventListener('mousemove', (e) => {
  const rect = button.getBoundingClientRect();
  const center = { x: rect.left + rect.width / 2, y: rect.top + rect.height / 2 };
  const { x, y } = magneticOffset({ x: e.clientX, y: e.clientY }, center, 60, 0.4);
  button.style.transform = `translate(${x}px, ${y}px)`;
});
button.addEventListener('mouseleave', () => {
  button.style.transform = 'translate(0, 0)';
});

3. Tilt-on-move card

Move your mouse across the card — it tilts toward your pointer in 3D.

Tilt me
View source 
function tiltAngles(pointerPos, rect, maxTiltDegrees) {
  const px = clamp((pointerPos.x - rect.left) / rect.width, 0, 1);
  const py = clamp((pointerPos.y - rect.top) / rect.height, 0, 1);
  const normX = px * 2 - 1;
  const normY = py * 2 - 1;
  return { rotateX: -normY * maxTiltDegrees, rotateY: normX * maxTiltDegrees };
}

card.addEventListener('mousemove', (e) => {
  const rect = card.getBoundingClientRect();
  const { rotateX, rotateY } = tiltAngles({ x: e.clientX, y: e.clientY }, rect, 12);
  card.style.transform = `perspective(600px) rotateX(${rotateX}deg) rotateY(${rotateY}deg)`;
});
card.addEventListener('mouseleave', () => {
  card.style.transform = 'perspective(600px) rotateX(0deg) rotateY(0deg)';
});

4. Spotlight card

A radial-gradient highlight follows your cursor across the card.

Move over me
View source 
/* CSS */
.spotlight-card {
  --x: 50%;
  --y: 50%;
  background: radial-gradient(circle 120px at var(--x) var(--y), rgba(124,159,255,.35), transparent 70%);
}

/* JS */
card.addEventListener('mousemove', (e) => {
  const rect = card.getBoundingClientRect();
  card.style.setProperty('--x', `${e.clientX - rect.left}px`);
  card.style.setProperty('--y', `${e.clientY - rect.top}px`);
});

5. Text-scramble on hover

Hover (or focus) the text below to decode it from scrambled characters.

HOVER TO DECODE
View source 
function scrambleStep(target, revealCount, random = Math.random) {
  let result = target.slice(0, revealCount);
  for (let i = revealCount; i < target.length; i++) {
    result += target[i] === ' ' ? ' ' : CHARS[Math.floor(random() * CHARS.length)];
  }
  return result;
}

el.addEventListener('mouseenter', () => {
  let revealCount = 0;
  const id = setInterval(() => {
    el.textContent = scrambleStep(target, revealCount++);
    if (revealCount > target.length) clearInterval(id);
  }, 30);
});

6. Cursor trail

Move your mouse inside the box to leave a fading particle trail.

View source 
let lastSpawn = 0;
container.addEventListener('mousemove', (e) => {
  const now = performance.now();
  if (now - lastSpawn < 40) return;
  lastSpawn = now;
  const rect = container.getBoundingClientRect();
  const particle = document.createElement('span');
  particle.className = 'trail-particle';
  particle.style.left = `${e.clientX - rect.left}px`;
  particle.style.top = `${e.clientY - rect.top}px`;
  container.appendChild(particle);
  setTimeout(() => particle.remove(), 600);
});
